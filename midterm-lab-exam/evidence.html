const checkoutForm = document.querySelector(".checkout-form");

checkoutForm.addEventListener("submit", (event) => {
  event.preventDefault();
  const deliveryDate = new FormData(checkoutForm).get("delivery-date");
  const confirmationUrl = new URL("order-confirmation.html", window.location.href);
  confirmationUrl.searchParams.set("delivery-date", deliveryDate);
  localStorage.removeItem("threadStoreCart");
  window.location.assign(confirmationUrl);
});

const dateValue = new URLSearchParams(window.location.search).get("delivery-date");

if (dateValue && /^\d{4}-\d{2}-\d{2}$/.test(dateValue)) {
  const parsedDate = new Date(`${dateValue}T00:00:00`);
  if (!Number.isNaN(parsedDate.getTime())) {
    deliveryDate.textContent = new Intl.DateTimeFormat("en-US", {

script.js

const cartItems = [];
const productLimit = 5;
let nextCartItemId = 1;

const cartLink = document.querySelector(".cart-link");
const miniCart = document.querySelector("#mini-cart");
const miniCartItems = document.querySelector(".mini-cart-items");
const emptyMessage = document.querySelector(".mini-cart-empty");
const cartCount = document.querySelector(".cart-count");
const cartTotal = document.querySelector("#cart-total");

function renderCart() {
  miniCartItems.replaceChildren();
  cartItems.forEach((item) => {
    const row = document.createElement("li");
    row.className = "mini-cart-item";
    const name = document.createElement("span");
    name.textContent = item.name;
    const price = document.createElement("strong");
    price.textContent = `$${item.price.toFixed(2)}`;
    const removeButton = document.createElement("button");
    removeButton.type = "button";
    removeButton.dataset.itemId = item.id;
    removeButton.textContent = "Remove";
    row.append(name, price, removeButton);
    miniCartItems.append(row);
  });
  const total = cartItems.reduce((sum, item) => sum + item.price, 0);
  cartCount.textContent = cartItems.length;
  cartTotal.textContent = `$${total.toFixed(2)}`;
  emptyMessage.hidden = cartItems.length > 0;
  document.querySelectorAll(".item").forEach((card) => {
    const productName = card.querySelector("h3").textContent.trim();
    const addButton = card.querySelector(".shop-btn");
    const quantity = cartItems.filter((item) => item.name === productName).length;
    addButton.disabled = quantity >= productLimit;
    addButton.textContent = addButton.disabled ? "Limit reached" : "Add to Cart";
  });
}

document.querySelectorAll(".item").forEach((card) => {
  const addButton = card.querySelector(".shop-btn");
  addButton.addEventListener("click", () => {
    const name = card.querySelector("h3").textContent.trim();
    const priceText = card.querySelector(".price").textContent;
    const price = Number(priceText.replace(/[^0-9.]/g, ""));
    const quantity = cartItems.filter((item) => item.name === name).length;
    if (quantity < productLimit) {
      cartItems.push({ id: nextCartItemId, name, price });
      nextCartItemId += 1;
      renderCart();
    }
  });
});

cartLink.addEventListener("click", (event) => {
  event.preventDefault();
  miniCart.hidden = !miniCart.hidden;
  cartLink.setAttribute("aria-expanded", String(!miniCart.hidden));
});

miniCartItems.addEventListener("click", (event) => {
  const removeButton = event.target.closest("button[data-item-id]");
  if (removeButton) {
    const itemIndex = cartItems.findIndex((item) => item.id === Number(removeButton.dataset.itemId));
    if (itemIndex !== -1) {
      cartItems.splice(itemIndex, 1);
      renderCart();
    }
  }
});

renderCart();
